---
title: "Create Assessment"
api: "POST /api/v1/assessments"
description: "Create a WhiteCollar assessment."
---

## Overview

Creates an assessment.

<Note>
  Requires `Content-Type: application/json` and a valid bearer access token. Use `test` mode while building integrations.
</Note>

---

## Request Body

<ParamField body="orgId" type="string" required>
  Organization identifier.
</ParamField>

<ParamField body="name" type="string" required>
  Assessment name.
</ParamField>

<ParamField body="assessmentType" type="string" required>
  Assessment type, such as `AI Powered Assessment` or `Customized Assessment`.
</ParamField>

<ParamField body="creationMethod" type="string" required>
  Creation method. Must be `ai` or `custom`.
</ParamField>

<ParamField body="mode" type="string" required>
  Environment scope. Must be `test` or `live`.
</ParamField>

<ParamField body="jobTitle" type="string" required>
  Target role title.
</ParamField>

<ParamField body="jobDescription" type="string" required>
  Job description.
</ParamField>

<ParamField body="roleType" type="string" required>
  Role type. Must be `IT` or `NON-IT`.
</ParamField>

<ParamField body="experienceRange" type="string" required>
  Experience range, such as `1-3 years`.
</ParamField>

<ParamField body="skills" type="array" required>
  Up to 5 skill identifiers.
</ParamField>

<ParamField body="totalMarks" type="number" required>
  Total assessment marks.
</ParamField>

<ParamField body="passMarks" type="number" required>
  Minimum passing marks.
</ParamField>

<ParamField body="duration" type="number" required>
  Time limit in minutes.
</ParamField>

<ParamField body="difficulty" type="string" required>
  Difficulty level. Must be `Easy`, `Medium`, `Hard`, or `Expert`.
</ParamField>

---

## Response Fields

<ResponseField name="success" type="boolean">
  `true` when the assessment is created.
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="Assessment fields">
    <ResponseField name="assessmentId" type="string">
      Unique assessment identifier.
    </ResponseField>
    <ResponseField name="name" type="string">
      Assessment name.
    </ResponseField>
    <ResponseField name="assessmentType" type="string">
      Assessment type.
    </ResponseField>
    <ResponseField name="creationMethod" type="string">
      Creation method.
    </ResponseField>
    <ResponseField name="mode" type="string">
      `test` or `live`.
    </ResponseField>
    <ResponseField name="roleType" type="string">
      `IT` or `NON-IT`.
    </ResponseField>
    <ResponseField name="isPublished" type="boolean">
      Whether the assessment is published.
    </ResponseField>
    <ResponseField name="isActive" type="boolean">
      Whether the assessment accepts candidates.
    </ResponseField>
    <ResponseField name="createdAt" type="string">
      ISO 8601 creation timestamp.
    </ResponseField>
  </Expandable>
</ResponseField>
